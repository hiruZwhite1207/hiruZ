-- Helper function để teleport giữa các sea
local function teleportToSea(n)
    local positions = {
        [2] = Vector3.new(1000, 0, 2000),
        [3] = Vector3.new(-500, 0, -500)
    }
    local pos = positions[n]
    if pos then
        local char = game.Players.LocalPlayer.Character
        if char then char.HumanoidRootPart.CFrame = CFrame.new(pos) end
    end
end
