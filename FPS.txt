local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")

-- 1. KHỞI TẠO TÙY CHỈNH GUI (Nền đen trong suốt, Di chuyển được)
local ScreenGui = Instance.new("ScreenGui", CoreGui)
ScreenGui.Name = "HiruzFPSCounterClassic"

local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 85, 0, 35) 
MainFrame.Position = UDim2.new(0.85, 0, 0.05, 0)
MainFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 15) -- Nền Đen
MainFrame.BackgroundTransparency = 0.4 -- Độ trong suốt của nền
MainFrame.BorderSizePixel = 0
MainFrame.Active = true
MainFrame.Draggable = true 

-- Bo góc cho bảng
local UICorner = Instance.new("UICorner", MainFrame)
UICorner.CornerRadius = UDim.new(0, 8)

-- Tạo viền trắng xung quanh bảng
local UIStroke = Instance.new("UIStroke", MainFrame)
UIStroke.Color = Color3.fromRGB(255, 255, 255) -- Viền Trắng
UIStroke.Thickness = 1.5 -- Độ dày viền
UIStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border

-- 2. CHỮ HIỂN THỊ ĐỊNH DẠNG (Chữ trắng nổi bật trên nền đen)
local FpsLabel = Instance.new("TextLabel", MainFrame)
FpsLabel.Size = UDim2.new(1, 0, 1, 0)
FpsLabel.BackgroundTransparency = 1
FpsLabel.Text = "FPS --" 
FpsLabel.TextColor3 = Color3.fromRGB(255, 255, 255) -- Chữ Trắng
FpsLabel.Font = Enum.Font.GothamBold
FpsLabel.TextSize = 13

-- 3. LOGIC ĐẾM FPS (Đã tối ưu khoảng thời gian lấy mẫu về 0.3s)
local fpsCount = 0
local lastUpdate = os.clock()

RunService.RenderStepped:Connect(function()
    fpsCount = fpsCount + 1
    local now = os.clock()
    
    -- Chỉnh từ 0.5/0.1 về 0.3 để số nhảy mượt, vừa mắt và chính xác nhất
    if now - lastUpdate >= 0.3 then
        local currentFps = math.floor(fpsCount / (now - lastUpdate))
        FpsLabel.Text = "FPS " .. tostring(currentFps)
        
        fpsCount = 0
        lastUpdate = now
    end
end)
