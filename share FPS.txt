local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")

-- 1. KHỞI TẠO TÙY CHỈNH GUI (Nền đỏ trong suốt, di chuyển được)
local ScreenGui = Instance.new("ScreenGui", CoreGui)
ScreenGui.Name = "HiruzFPSCounterClean"

local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 85, 0, 35) -- Thu nhỏ lại cho vừa khít với chữ ngắn
MainFrame.Position = UDim2.new(0.85, 0, 0.05, 0)
MainFrame.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
MainFrame.BackgroundTransparency = 0.4
MainFrame.BorderSizePixel = 0
MainFrame.Active = true
MainFrame.Draggable = true 

local UICorner = Instance.new("UICorner", MainFrame)
UICorner.CornerRadius = UDim.new(0, 8)

-- 2. CHỮ HIỂN THỊ ĐỊNH DẠNG: FPS 120
local FpsLabel = Instance.new("TextLabel", MainFrame)
FpsLabel.Size = UDim2.new(1, 0, 1, 0)
FpsLabel.BackgroundTransparency = 1
FpsLabel.Text = "FPS --" -- Định dạng mới không dấu hai chấm
FpsLabel.TextColor3 = Color3.fromRGB(0, 0, 0)
FpsLabel.Font = Enum.Font.GothamBold
FpsLabel.TextSize = 14

-- 3. LOGIC ĐẾM FPS BẰNG DELTA X
local fpsCount = 0
local lastUpdate = os.clock()

RunService.RenderStepped:Connect(function()
    fpsCount = fpsCount + 1
    local now = os.clock()
    
    -- Cập nhật mỗi 0.1 giây
    if now - lastUpdate >= 0.1 then
        local currentFps = math.floor(fpsCount / (now - lastUpdate))
        FpsLabel.Text = "FPS " .. tostring(currentFps) -- Hiển thị dạng "FPS 120"
        
        fpsCount = 0
        lastUpdate = now
    end
end)
